'''
#include <SoftwareSerial.h>

// 블루투스 핀 설정 (RX: D2, TX: D3)
// ⚠️ HC-06의 TXD -> 아두이노 D2 / HC-06의 RXD -> 아두이노 D3
SoftwareSerial BTSerial(2, 3); 

const int MOTOR_1 = 4;     // 디지털 모터 핀
const int MOTOR_2 = 5;     // PWM 모터 핀 (진동 세기 변경용)
const int HEATER_PIN = 6;  // PWM 온열 패드 핀

// 가동 제어를 위한 전역 변수
char current_emotion = 'N'; 

void setup() {
  pinMode(MOTOR_1, OUTPUT);
  pinMode(MOTOR_2, OUTPUT);
  pinMode(HEATER_PIN, OUTPUT);
  
  Serial.begin(9600);   
  BTSerial.begin(9600); 
  Serial.println("=== 감정별 하드웨어 동작 테스트 시작 ===");
}

void loop() {
  // 1. Flask 서버(블루투스)로부터 신호가 들어오면 현재 감정 상태를 갱신
  if (BTSerial.available()) {
    current_emotion = BTSerial.read();
    Serial.print("수신된 감정 신호: ");
    Serial.println(current_emotion);                                                                                                                                                                                                                                                                                                                                        
    
    // 신호가 바뀔 때 오작동 방지를 위해 일단 모두 끄고 시작
    digitalWrite(MOTOR_1, LOW);
    analogWrite(MOTOR_2, 0);
    analogWrite(HEATER_PIN, 0);
  }

  // 2. 현재 감정 상태에 따라 다르게 작동 (동작 확인용 단순 패턴)
  if (current_emotion == '0' || current_emotion == 'N') {
    // [평온 / 정지] -> 모두 끔
    digitalWrite(MOTOR_1, LOW);
    analogWrite(MOTOR_2, 0);
    analogWrite(HEATER_PIN, 0);
  } 
  
  else if (current_emotion == 'A') {
    // [불안] -> 최고 온도, 미치도록 빠른 진동 (쿵쾅쿵쾅)
    analogWrite(HEATER_PIN, 240); // 높은 온도
    
    digitalWrite(MOTOR_1, HIGH); analogWrite(MOTOR_2, 255); delay(100);
    digitalWrite(MOTOR_1, LOW);  analogWrite(MOTOR_2, 0);   delay(100);
  } 
  
  else if (current_emotion == 'S') {
    // [슬픔] -> 미지근한 온도, 아주 느리고 긴 진동 (쿠웅-------쿠웅)
    analogWrite(HEATER_PIN, 150); // 낮은 온도
    
    digitalWrite(MOTOR_1, HIGH); analogWrite(MOTOR_2, 130); delay(400);
    digitalWrite(MOTOR_1, LOW);  analogWrite(MOTOR_2, 0);   delay(800);
  } 
  
  else if (current_emotion == 'R') {
    // [분노] -> 한계치 최대 온도, 우렁차고 강한 진동 (쾅! 쾅! 쾅!)
    analogWrite(HEATER_PIN, 255); // 최대 온도
    
    digitalWrite(MOTOR_1, HIGH); analogWrite(MOTOR_2, 255); delay(200);
    digitalWrite(MOTOR_1, LOW);  analogWrite(MOTOR_2, 0);   delay(300);
  } 
  
  else if (current_emotion == 'H') {
    // [행복] -> 적당히 따뜻한 온도, 일정한 리듬의 진동 (둥-탁 둥-탁)
    analogWrite(HEATER_PIN, 180); // 중간 온도
    
    digitalWrite(MOTOR_1, HIGH); analogWrite(MOTOR_2, 180); delay(150);
    digitalWrite(MOTOR_1, LOW);  analogWrite(MOTOR_2, 0);   delay(400);
  }
}'''
