#include <SoftwareSerial.h>

// 블루투스 설정
SoftwareSerial BTSerial(2, 3); 

const int MOTOR_1 = 4;     // 디지털 핀 (주 모터)
const int MOTOR_2 = 5;     // PWM 핀 (보조 모터)
const int HEATER_PIN = 6;  // PWM 온열

struct HapticParams { float bpm; int pwm; int heat; };
const HapticParams SADNESS   = {60.0, 240, 250}; // 강도 상향
const HapticParams ANGER     = {75.0, 190, 180}; 
const HapticParams ANXIETY   = {85.0, 160, 200}; 
const HapticParams NEUTRAL   = {68.0, 110, 130}; 
const HapticParams HAPPINESS = {65.0, 110, 160}; 

char current_emotion = '0';
unsigned long last_beat_millis = 0;
int beat_phase = 0; 
float live_bpm = 68.0;
int live_pwm = 110;
int live_heat = 0;

void setup() {
  pinMode(MOTOR_1, OUTPUT);
  pinMode(MOTOR_2, OUTPUT);
  pinMode(HEATER_PIN, OUTPUT);
  Serial.begin(9600);   
  BTSerial.begin(9600); 
  stopAll();
}

void loop() {
  if (BTSerial.available()) {
    char cmd = BTSerial.read();
    handleCommand(cmd);
  }
  runHapticEngine();
}

void handleCommand(char cmd) {
  if (cmd == '0') { stopAll(); return; }
  if (cmd == '+') { live_heat = min(live_heat + 25, 255); analogWrite(HEATER_PIN, live_heat); return; }
  if (cmd == '-') { live_heat = max(live_heat - 25, 0);   analogWrite(HEATER_PIN, live_heat); return; }

  current_emotion = cmd;
  switch (cmd) {
    case 'S': case 's': applyParams(SADNESS);   break;
    case 'R': case 'r': applyParams(ANGER);     break;
    case 'A': case 'a': applyParams(ANXIETY);   break;
    case 'N': case 'n': applyParams(NEUTRAL);   break;
    case 'H': case 'h': applyParams(HAPPINESS); break;
  }
}

void applyParams(HapticParams p) {
  live_bpm = p.bpm; live_pwm = p.pwm; live_heat = p.heat;
  analogWrite(HEATER_PIN, live_heat);
  beat_phase = 0;
}

void runHapticEngine() {
  if (current_emotion == '0') return;
  unsigned long now = millis();
  unsigned long full_interval = (60.0 / live_bpm) * 1000;

  switch (beat_phase) {
    case 0: // 대기
      if (now - last_beat_millis >= full_interval) {
        if (current_emotion == 'N' && random(0, 4) != 0) { last_beat_millis = now; return; }
        startLub(); // 1번+2번 동시 가동 (강한 쿵)
        last_beat_millis = now;
        beat_phase = 1;
      }
      break;
    case 1: // Lub 지속 (90ms)
      if (now - last_beat_millis >= 90) {
        stopMotors();
        last_beat_millis = now;
        beat_phase = 2;
      }
      break;
    case 2: // 휴지 (110ms)
      if (now - last_beat_millis >= 110) {
        startDub(); // 2번만 가동 (약한 쾅)
        last_beat_millis = now;
        beat_phase = 3;
      }
      break;
    case 3: // Dub 지속 (70ms)
      if (now - last_beat_millis >= 70) {
        stopMotors();
        last_beat_millis = now;
        beat_phase = 0;
      }
      break;
  }
}

void startLub() {
  // 제1심음: 묵직한 타격감을 위해 두 모터를 모두 켭니다.
  digitalWrite(MOTOR_1, HIGH);
  analogWrite(MOTOR_2, live_pwm);
}

void startDub() {
  // 제2심음: 여운을 남기기 위해 PWM 모터(2번)만 사용합니다.
  digitalWrite(MOTOR_1, LOW);
  analogWrite(MOTOR_2, live_pwm * 0.7); 
}

void stopMotors() {
  digitalWrite(MOTOR_1, LOW);
  analogWrite(MOTOR_2, 0);
}

void stopAll() {
  current_emotion = '0';
  stopMotors();
  analogWrite(HEATER_PIN, 0);
  beat_phase = 0;
}
